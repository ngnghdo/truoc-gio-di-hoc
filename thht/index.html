Bài 1. Tín hiệu liên tục

I. Hàm bước nhảy u(t) & hàm dốc r(t)

Bài 1–2: viết ustep, uramp · Bài 3: vẽ 5u(t−2), 3r(t+5), 2r(t+2,5)−5r(t)+3r(t−2)+u(t−4), sin(t)[u(t+3)−u(t−3)] · Bài 4: dựng lại 2 hình mẫu.

bai1_1.m
%% ===== BÀI 1 – I. HÀM BƯỚC NHẢY u(t) VÀ HÀM DỐC r(t) (Bài 1 → Bài 4) =====
clc; clear; close all;% clc: xóa chữ trong Command Window | clear: xóa mọi biến cũ | close all: đóng mọi cửa sổ hình
t = -10:0.01:10;% vector thời gian: bắt đầu -10, bước 0.01, kết thúc 10 → 2001 điểm; bước nhỏ nên đồ thị trông liên tục
% ---------- Bài 3: vẽ 4 tín hiệu dùng ustep, uramp (2 hàm tự viết ở CUỐI file) ----------
y1 = 5*ustep(t - 2);% 5u(t-2): (t-2) dịch u(t) sang PHẢI 2 → nhảy tại t=2; nhân 5 → bậc cao 5
y2 = 3*uramp(t + 5);% 3r(t+5): (t+5) dịch r(t) sang TRÁI 5 → bắt đầu dốc tại t=-5; nhân 3 → độ dốc 3
y3 = 2*uramp(t + 2.5) - 5*uramp(t) + 3*uramp(t - 2) + ustep(t - 4);% cộng/trừ từng mảnh: mỗi r(t-a) đổi độ dốc tại t=a, u(t-4) nhảy +1 tại t=4
y4 = sin(t) .* (ustep(t + 3) - ustep(t - 3));% u(t+3)-u(t-3) = "cửa sổ" bằng 1 trên [-3,3], 0 ngoài; .* nhân TỪNG PHẦN TỬ → chỉ giữ sin(t) trong [-3,3]
figure('Name', 'Bai 1 - I - Bai 3');% mở cửa sổ hình mới, 'Name' = tên hiện trên thanh tiêu đề cửa sổ
subplot(2,2,1);% chia cửa sổ thành lưới 2 hàng × 2 cột, vẽ vào ô số 1 (đếm trái→phải, trên→dưới)
plot(t, y1, 'LineWidth', 1.5);% vẽ y1 theo t; 'LineWidth',1.5 → nét dày 1.5 pt
grid on; ylim([-1 6]);% grid on: bật lưới | ylim([a b]): cố định trục y từ a đến b cho dễ nhìn bước nhảy
title('5u(t-2)'); xlabel('t'); ylabel('y(t)');% tiêu đề ô + nhãn trục x, trục y
subplot(2,2,2);% ô số 2 (hàng 1, cột 2)
plot(t, y2, 'LineWidth', 1.5); grid on;% vẽ y2, bật lưới
title('3r(t+5)'); xlabel('t'); ylabel('y(t)');% tiêu đề + nhãn 2 trục
subplot(2,2,3);% ô số 3 (hàng 2, cột 1)
plot(t, y3, 'LineWidth', 1.5); grid on;% vẽ y3, bật lưới
title('2r(t+2.5) - 5r(t) + 3r(t-2) + u(t-4)'); xlabel('t'); ylabel('y(t)');% tiêu đề ô 3 + nhãn trục
subplot(2,2,4);% ô số 4 (hàng 2, cột 2)
plot(t, y4, 'LineWidth', 1.5); grid on; ylim([-1.2 1.2]);% vẽ y4 | ylim: trục y từ -1.2 đến 1.2 (biên độ sin ≤ 1)
title('sin(t)[u(t+3) - u(t-3)]'); xlabel('t'); ylabel('y(t)');% tiêu đề ô 4 + nhãn trục
% ---------- Bài 4: dựng lại 2 hình mẫu từ u(t), r(t) ----------
% Hình 1: tam giác đáy [-4,4], đỉnh cao 2 tại t=0 → dốc lên 2/4 = 0.5 rồi dốc xuống -0.5
z1 = 0.5*uramp(t + 4) - uramp(t) + 0.5*uramp(t - 4);% tại -4: độ dốc 0→+0.5 | tại 0: +0.5→-0.5 (trừ 1·r) | tại 4: -0.5→0 (cộng 0.5·r)
% Hình 2: bậc cao 1 trên [-8,8] + tam giác đáy [-4,4] cao thêm 2 (đỉnh tổng = 3)
z2 = ustep(t + 8) - ustep(t - 8) + z1;% u(t+8)-u(t-8): xung chữ nhật cao 1 trên [-8,8]; cộng z1 → đỉnh 1+2 = 3
figure('Name', 'Bai 1 - I - Bai 4');% cửa sổ hình mới cho Bài 4
subplot(2,1,1);% lưới 2 hàng × 1 cột, ô 1 (trên)
plot(t, z1, 'LineWidth', 1.5); grid on; axis([-10 10 -0.5 2]);% axis([xmin xmax ymin ymax]): đặt cả 2 trục giống hình mẫu
title('Hình 1: 0.5r(t+4) - r(t) + 0.5r(t-4)');% tiêu đề ô trên
subplot(2,1,2);% ô 2 (dưới)
plot(t, z2, 'LineWidth', 1.5); grid on; axis([-10 10 0 3]);% vẽ z2 | trục giống hình mẫu 2: t ∈ [-10,10], y ∈ [0,3]
title('Hình 2: u(t+8) - u(t-8) + 0.5r(t+4) - r(t) + 0.5r(t-4)');% tiêu đề ô dưới
% ---------- Bài 1 + Bài 2: định nghĩa hàm (hàm cục bộ phải đặt CUỐI script, MATLAB R2016b trở lên) ----------
function y = ustep(t)% khai báo hàm: tên ustep, đầu vào t (số hoặc vector), đầu ra y cùng kích thước với t
y = double(t >= 0);% t>=0 cho mảng logic (1 nếu đúng, 0 nếu sai); double() đổi sang kiểu số để nhân/cộng được
end% kết thúc hàm ustep
function y = uramp(t)% hàm dốc đơn vị: r(t) = t khi t ≥ 0, = 0 khi t < 0
y = t .* (t >= 0);% t .* (t>=0): nhân từng phần tử t với 1 (t≥0) hoặc 0 (t<0) → giữ t ở nửa dương, còn lại bằng 0
end% kết thúc hàm uramp (cách khác tương đương: y = max(t, 0);)

II. Phần chẵn, phần lẻ

Hàm evenodd + vẽ y(t)=2r(t+2,5)−5r(t)+3r(t−2)+u(t−4) cùng phần chẵn, phần lẻ trên 1 đồ thị.

bai1_2.m
%% ===== BÀI 1 – II. PHẦN CHẴN, PHẦN LẺ CỦA TÍN HIỆU =====
clc; clear; close all;% xóa màn hình lệnh, xóa biến, đóng hình cũ
t = -10:0.01:10;% t PHẢI đối xứng quanh 0 (từ -a đến a, cùng bước) thì fliplr(y) mới đúng là y(-t)
y = 2*uramp(t + 2.5) - 5*uramp(t) + 3*uramp(t - 2) + ustep(t - 4);% tín hiệu đề cho, dùng lại ustep/uramp ở cuối file
[ye, yo] = evenodd(y);% gọi hàm evenodd: trả về 2 đầu ra cùng lúc → ye = phần chẵn, yo = phần lẻ
figure('Name', 'Bai 1 - II');% cửa sổ hình mới
plot(t, y, 'k-', 'LineWidth', 2); hold on;% 'k-': màu đen (k), nét liền (-) | hold on: giữ hình để vẽ chồng đường sau
plot(t, ye, 'r--', 'LineWidth', 1.5);% 'r--': màu đỏ, nét đứt dài → phần chẵn
plot(t, yo, 'b-.', 'LineWidth', 1.5); hold off;% 'b-.': màu xanh dương, nét gạch-chấm → phần lẻ | hold off: thôi vẽ chồng
grid on; xlabel('t'); ylabel('Biên độ');% bật lưới + nhãn 2 trục
legend('y(t)', 'y_e(t) – phần chẵn', 'y_o(t) – phần lẻ', 'Location', 'best');% chú thích theo đúng thứ tự 3 lệnh plot; 'best': MATLAB tự chọn góc ít che đồ thị
title('y(t) = 2r(t+2.5) - 5r(t) + 3r(t-2) + u(t-4) và phần chẵn/lẻ');% tiêu đề hình
% Kiểm tra nhanh (không bắt buộc): ye + yo phải trả lại đúng y
disp(max(abs(ye + yo - y)));% sai lệch lớn nhất giữa ye+yo và y; in ra 0 (hoặc ~1e-15) là đúng
% ---------- Các hàm (đặt cuối script) ----------
function [ye, yo] = evenodd(y)% hàm 1 đầu vào (y), 2 đầu ra (ye, yo) — y là vector mẫu tín hiệu trên t đối xứng
yr = fliplr(y);% fliplr: lật vector trái↔phải → mẫu tại t thành mẫu tại -t, tức yr = y(-t)
ye = 0.5*(y + yr);% phần chẵn: ye(t) = [y(t) + y(-t)]/2 → thỏa ye(-t) = ye(t)
yo = 0.5*(y - yr);% phần lẻ: yo(t) = [y(t) - y(-t)]/2 → thỏa yo(-t) = -yo(t)
end% (chú ý: trong PDF dấu trừ bị in thành "–" dài, gõ lại bằng "-" thường kẻo lỗi)
function y = ustep(t)% u(t): 1 khi t ≥ 0, 0 khi t < 0
y = double(t >= 0);% so sánh → logic 0/1 → đổi sang số
end% kết thúc hàm ustep
function y = uramp(t)% r(t): t khi t ≥ 0, 0 khi t < 0
y = t .* (t >= 0);% nhân từng phần tử với mặt nạ 0/1
end% kết thúc hàm uramp

III. Tổng các tín hiệu tuần hoàn

Vẽ x₁, x₂ trên [−10, 10], xét tuần hoàn, tìm chu kỳ.

bai1_3.m
%% ===== BÀI 1 – III. TỔNG CÁC TÍN HIỆU TUẦN HOÀN =====
clc; clear; close all;% dọn màn hình, biến, hình
t = -10:0.001:10;% bước 0.001 (mịn hơn) vì cos(6πt) dao động nhanh, chu kỳ chỉ 1/3 s
W0 = pi/10;% Ω0 = π/10 (rad/s)
x1 = 1 + 1.5*cos(2*pi*W0*t) - 0.6*cos(4*W0*t);% a. x1(t): cos(...) nhận vector t → trả vector, cộng/nhân số áp cho mọi phần tử
x2 = 1 + 1.5*cos(6*pi*t) - 0.6*cos(4*W0*t);% b. x2(t)
figure('Name', 'Bai 1 - III - Tuan hoan');% mở cửa sổ hình mới, đặt tên
subplot(2,1,1);% ô trên: x1
plot(t, x1, 'LineWidth', 1.2); grid on; xlabel('t'); ylabel('x_1(t)');% dấu _ trong nhãn → chữ số 1 viết chỉ số dưới
title('a. x_1(t) = 1 + 1.5cos(2\pi\Omega_0t) - 0.6cos(4\Omega_0t)');% \pi, \Omega: MATLAB hiển thị thành ký hiệu π, Ω
subplot(2,1,2);% ô dưới: x2
plot(t, x2, 'LineWidth', 1.2); grid on; xlabel('t'); ylabel('x_2(t)');% vẽ x2, bật lưới, nhãn trục (x_2 → chỉ số dưới)
title('b. x_2(t) = 1 + 1.5cos(6\pit) - 0.6cos(4\Omega_0t)');% tiêu đề; \pi → π, \Omega_0 → Ω₀
% ---------- Tìm chu kỳ: tổng tuần hoàn ⇔ tỉ số các chu kỳ thành phần là số HỮU TỈ ----------
% Thành phần cos(w·t) có chu kỳ T = 2π/w
T1a = 2*pi/(2*pi*W0);% x1, số hạng 1: w = 2πΩ0 = π²/5 → T = 10/π ≈ 3.18 s
T2a = 2*pi/(4*W0);% x1, số hạng 2: w = 4Ω0 = 2π/5 → T = 5 s
fprintf('x1: T1 = %.4f, T2 = %.4f, T1/T2 = %.4f\n', T1a, T2a, T1a/T2a);% %.4f: in số thực 4 chữ số thập phân | \n: xuống dòng
% → T1/T2 = 2/π là số VÔ TỈ ⇒ x1(t) KHÔNG tuần hoàn (đồ thị không lặp lại đúng hình dạng)
T1b = 2*pi/(6*pi);% x2, số hạng 1: w = 6π → T = 1/3 s
T2b = 2*pi/(4*W0);% x2, số hạng 2: T = 5 s
fprintf('x2: T1 = %.4f, T2 = %.4f, T2/T1 = %.4f\n', T1b, T2b, T2b/T1b);% in 3 số: T1, T2 và tỉ số T2/T1 của x2
% → T2/T1 = 15 là số HỮU TỈ ⇒ x2(t) TUẦN HOÀN, chu kỳ T = BCNN(1/3, 5) = 5 s (= 15·T1 = 1·T2)
hold on;% đang ở ô dưới (x2): vẽ thêm đường dọc đánh dấu các mốc cách nhau 1 chu kỳ
xline([-10 -5 0 5 10], 'r--');% xline: vẽ đường thẳng đứng tại các giá trị t đã cho (MATLAB R2018b+) → thấy đồ thị lặp lại sau mỗi 5 s
hold off;% thôi vẽ chồng

III. Năng lượng, công suất

x(t)=e^(−t)cos(2πt)u(t) trên [−10, 10], dùng biến tượng trưng (syms).

bai1_4.m
%% ===== BÀI 1 – III. NĂNG LƯỢNG, CÔNG SUẤT CỦA TÍN HIỆU (biến tượng trưng) =====
% x(t) = e^(-t)·cos(2πt)·u(t) trên đoạn -10 ≤ t ≤ 10 → T = 20, [-T/2, T/2] = [-10, 10]
% E = ∫ |x(t)|² dt (từ -T/2 đến T/2) P = E / T
clc; clear; close all;% dọn màn hình, biến, hình
syms t real% khai báo t là biến TƯỢNG TRƯNG (không cần giá trị số); 'real' → t là số thực, |x|² = x²
x = exp(-t)*cos(2*pi*t);% biểu thức tượng trưng phần t ≥ 0 của x(t) (dùng * thường vì đây là ký hiệu, không phải vector)
T = 20;% độ dài đoạn xét: từ -10 đến 10
% u(t) = 0 khi t < 0 ⇒ |x(t)|² = 0 trên [-10, 0) ⇒ cận dưới tích phân thực chất là 0
E = int(x^2, t, 0, T/2);% int(biểu thức, biến, cận dưới, cận trên): tích phân xác định ∫₀¹⁰ e^(-2t)cos²(2πt) dt
E = simplify(E);% rút gọn biểu thức kết quả cho gọn
Ed = double(E);% double(): đổi kết quả tượng trưng (chứa π, exp) thành số thực để đọc giá trị
P = E / T;% công suất trung bình P = E/T (vẫn dạng tượng trưng)
Pd = double(P);% đổi P sang số
disp('Năng lượng E (dạng chính xác) ='); pretty(E);% pretty: in biểu thức toán dạng "2 chiều" dễ đọc
fprintf('E ≈ %.6f\n', Ed);% in E dạng số, 6 chữ số thập phân (≈ 0.256176)
fprintf('P ≈ %.6f\n', Pd);% in P dạng số (≈ 0.012809)
% Vẽ x(t) đầy đủ trên [-10, 10]: nhân thêm hàm bước tượng trưng heaviside(t) (= u(t))
xu = x*heaviside(t);% heaviside(t): 0 khi t < 0, 1 khi t > 0 → x(t) bằng 0 ở nửa âm như đề cho
fplot(xu, [-10 10], 'LineWidth', 1.3);% fplot: vẽ trực tiếp biểu thức tượng trưng trên khoảng [-10, 10] (ezplot là lệnh cũ)
grid on; xlabel('t'); ylabel('x(t)');% bật lưới + nhãn 2 trục
title('x(t) = e^{-t}cos(2\pit)u(t)');% ^{...}: số mũ nhiều ký tự trong tiêu đề

IV. Dịch, co giãn, đảo

x(t)=e^(−|t|): x(t±2) · x(2t), x(0,5t) · x(−t).

bai1_5.m
%% ===== BÀI 1 – IV. PHÉP DỊCH, CO GIÃN, ĐẢO TÍN HIỆU x(t) = e^(-|t|) =====
clc; clear; close all;% dọn màn hình, biến, hình
t = -10:0.01:10;% trục thời gian chung
x = @(t) exp(-abs(t));% HÀM ẨN DANH: @(t) = "hàm theo biến t"; gọi x(t-2), x(2*t)... như viết toán | abs: trị tuyệt đối
% ---------- Bài 1: dịch — x(t), x(t-2), x(t+2) ----------
figure('Name', 'Bai 1 - IV - Dich');% cửa sổ hình 1: phép dịch
plot(t, x(t), 'k-', 'LineWidth', 2); hold on;% gốc: đỉnh tại t = 0
plot(t, x(t - 2), 'r--', 'LineWidth', 1.5);% x(t-2): trễ → dịch sang PHẢI 2, đỉnh tại t = 2
plot(t, x(t + 2), 'b-.', 'LineWidth', 1.5); hold off;% x(t+2): sớm → dịch sang TRÁI 2, đỉnh tại t = -2
grid on; xlabel('t'); ylabel('Biên độ'); title('Phép dịch');% lưới, nhãn trục, tiêu đề
legend('x(t)', 'x(t-2)', 'x(t+2)');% chú thích theo thứ tự vẽ
% ---------- Bài 2: co giãn — x(t), x(2t), x(0.5t) ----------
figure('Name', 'Bai 1 - IV - Co gian');% cửa sổ hình 2: phép co giãn
plot(t, x(t), 'k-', 'LineWidth', 2); hold on;% gốc x(t): đen nét liền | hold on: giữ để vẽ chồng
plot(t, x(2*t), 'r--', 'LineWidth', 1.5);% x(2t): hệ số > 1 → NÉN theo trục t 2 lần (đồ thị hẹp lại)
plot(t, x(0.5*t), 'b-.', 'LineWidth', 1.5); hold off;% x(0.5t): hệ số < 1 → GIÃN 2 lần (đồ thị rộng ra)
grid on; xlabel('t'); ylabel('Biên độ'); title('Phép co giãn');% lưới, nhãn trục, tiêu đề
legend('x(t)', 'x(2t)', 'x(0.5t)');% chú thích 3 đường theo thứ tự vẽ
% ---------- Bài 3: đảo — x(t), x(-t) ----------
figure('Name', 'Bai 1 - IV - Dao');% cửa sổ hình 3: phép đảo
plot(t, x(t), 'k-', 'LineWidth', 2); hold on;% gốc x(t): đen nét liền, giữ hình
plot(t, x(-t), 'r--', 'LineWidth', 1.5); hold off;% x(-t): lật đối xứng qua trục tung
grid on; xlabel('t'); ylabel('Biên độ'); title('Phép đảo');% lưới, nhãn trục, tiêu đề
legend('x(t)', 'x(-t)');% chú thích 2 đường
% Nhận xét: e^(-|t|) là hàm CHẴN nên x(-t) trùng khít x(t) — 2 đường chồng lên nhau

Bài 2. Hàm tuyến tính

1. Tần số và nốt nhạc

Chơi + vẽ 1 nốt · chơi bản CCGGAAG--, FFEEDDC--.

bai2_1.m
%% ===== BÀI 2 – 1. TẦN SỐ VÀ NỐT NHẠC =====
% Mỗi nốt nhạc = 1 tín hiệu sin có tần số riêng. Bảng 1: C D E F G A B = 262 294 330 349 392 440 494 Hz
clc; clear; close all;% dọn màn hình, biến, hình
% ---------- A. Chơi và vẽ 1 nốt (ví dụ trong tài liệu: nốt B, 494 Hz) ----------
Fs = 8000;% tần số lấy mẫu: 8000 mẫu mỗi giây (đủ cho âm < 4000 Hz)
T = 2;% thời gian phát nốt: 2 giây
t = 0:1/Fs:T;% vector thời điểm lấy mẫu, cách nhau 1/Fs giây → 16001 phần tử
Amp = 1;% biên độ (âm lượng); sound() chỉ nhận giá trị trong [-1, 1]
ph = 0;% pha ban đầu (rad)
fb = 494;% tần số nốt B
x = Amp*sin(2*pi*fb*t + ph);% sin(2π·f·t + φ) tính cho MỌI phần tử của t → vector mẫu âm thanh
% thử âm "tắt dần" như phím đàn: x = Amp*exp(-t).*sin(2*pi*fb*t + ph); (.* vì nhân 2 vector từng phần tử)
figure('Name', 'Bai 2 - 1 not B');% mở cửa sổ hình, đặt tên
plot(t(1:300), x(1:300));% chỉ vẽ 300 mẫu đầu (≈ 0.0375 s ≈ 18 chu kỳ) — vẽ hết 16001 mẫu sẽ thành một vệt đặc
grid on; xlabel('t (s)'); ylabel('x(t)'); title('Nốt B – 494 Hz');% lưới, nhãn trục, tiêu đề
sound(x, Fs);% phát vector x ra loa với tốc độ Fs mẫu/giây
pause(T + 0.5);% sound() KHÔNG chờ phát xong → dừng chương trình T+0.5 s để nốt sau không đè lên
% ---------- B. Chơi bản nhạc: CCGGAAG--, FFEEDDC-- ----------
names = 'CDEFGAB';% chuỗi tên nốt, vị trí k trong chuỗi ↔ tần số thứ k ở dòng dưới
freqs = [262 294 330 349 392 440 494];% tần số (Hz) theo Bảng 1
song = 'CCGGAAG--FFEEDDC--';% bản nhạc: chữ cái = 1 nốt dài 1 phách; mỗi '-' = ngân nốt TRƯỚC đó thêm 1 phách
beat = 0.4;% độ dài 1 phách (giây) — chỉnh số này để nhạc nhanh/chậm
gap = zeros(1, round(0.03*Fs));% khoảng lặng 0.03 s giữa 2 nốt (vector toàn số 0) để nghe tách bạch 2 nốt giống nhau
% B1. Đọc chuỗi bản nhạc → danh sách nốt + số phách của từng nốt
noteList = '';% các nốt (chuỗi ký tự rỗng ban đầu)
beats = [];% số phách tương ứng (vector rỗng ban đầu)
for k = 1:length(song)% duyệt từng ký tự của chuỗi song
if song(k) == '-'% gặp dấu '-'
beats(end) = beats(end) + 1;% end = phần tử cuối → nốt vừa thêm được ngân thêm 1 phách
else% gặp chữ cái (một nốt mới)
noteList(end+1) = song(k);% end+1: thêm vào cuối chuỗi
beats(end+1) = 1;% nốt mới mặc định dài 1 phách
end% kết thúc if
end% kết thúc vòng for đọc chuỗi
% B2. Tạo âm cho từng nốt rồi nối lại thành 1 vector dài
y = [];% vector âm thanh của cả bài, ban đầu rỗng
for k = 1:length(noteList)% duyệt từng nốt
f = freqs(names == noteList(k));% names==... cho mảng logic 1 ở đúng vị trí tên nốt → lấy ra tần số tương ứng
d = beats(k)*beat;% thời lượng nốt = số phách × độ dài 1 phách
tn = 0:1/Fs:d;% trục thời gian riêng cho nốt này
xn = exp(-2*tn/d) .* sin(2*pi*f*tn);% sin tần số f, nhân bao hình e^(-2t/d) cho âm nhỏ dần (giống phím đàn); .* từng phần tử
y = [y, xn, gap];% nối ngang: [cũ, nốt mới, lặng] → y dài dần
end% kết thúc vòng for tạo âm
y = 0.9*y/max(abs(y));% chuẩn hóa biên độ về tối đa 0.9 (trong [-1,1]) để sound() không bị méo/cắt
sound(y, Fs);% phát cả bài
figure('Name', 'Bai 2 - Ban nhac');% cửa sổ hình cho cả bản nhạc
plot((0:length(y)-1)/Fs, y);% trục thời gian = chỉ số mẫu (0,1,2,...) / Fs
grid on; xlabel('t (s)'); ylabel('y(t)'); title(['Bản nhạc: ' song]);% ['a' b]: nối chuỗi để ghép tiêu đề

2. Chuỗi Fourier của kèn trumpet

%% ===== BÀI 2 – 2. CHUỖI FOURIER CỦA KÈN TRUMPET (nốt B, 494 Hz) =====
% x(t) = Σ_{k=1..9} A_k·cos(2π·k·494·t − θ_k) (A0 = 0: không có thành phần một chiều DC)
clc; clear; close all;% dọn màn hình, biến, hình
% ---------- (a) Tổng hợp tín hiệu từ 9 số hạng và nghe ----------
t = linspace(0, 1, 44100);% linspace(đầu, cuối, số điểm): 44100 điểm cách đều trong 1 s → 44100 mẫu/giây (như đề yêu cầu)
F = 494;% tần số cơ bản (nốt B); họa âm thứ k có tần số k·F
Fs = 44100;% tần số lấy mẫu tương ứng, dùng cho sound()
A = [0.1155 0.3417 0.1789 0.1232 0.0678 0.0473 0.0260 0.0065 0.0020];% biên độ A1..A9 (Bảng 2)
th = [-2.1299 1.6727 -2.5454 0.6607 -2.0390 2.1597 -1.0467 1.8581 -2.3925];% pha θ1..θ9 (rad) (Bảng 2, giữ đúng dấu ±)
x = zeros(size(t));% vector toàn 0 cùng kích thước t, dùng để cộng dồn
for k = 1:9% k = 1 (cơ bản), 2..9 (họa âm)
x = x + A(k)*cos(2*pi*k*F*t - th(k));% cộng thêm số hạng thứ k: biên độ A(k), tần số k·F, pha θ(k)
end% kết thúc vòng for
% (cách viết gọn không cần vòng lặp: x = A*cos(2*pi*F*(1:9)'*t - th'); — (1:9)'*t tạo ma trận 9×44100, A (1×9) nhân ma trận → cộng 9 hàng)
sound(x/max(abs(x)), Fs);% chuẩn hóa về [-1,1] rồi phát: nghe giống tiếng kèn hơn sin đơn
pause(1.5);% chờ phát xong (1 s + dư) trước khi phát bản (c)
% ---------- (b) Vẽ 3 chu kỳ: 3·T = 3/494 ≈ 6.1e-3 s (đề ghi ~4.5e-3 s; muốn đúng số đề thì thay 3/F bằng 4.5e-3) ----------
idx = t <= 3/F;% mảng logic: true tại các mẫu có t ≤ 3 chu kỳ → dùng làm "bộ lọc chỉ số"
figure('Name', 'Bai 2 - Trumpet');% mở cửa sổ hình, đặt tên
plot(t(idx)*1e3, x(idx), 'b', 'LineWidth', 1.5);% t(idx): chỉ lấy mẫu thỏa điều kiện | *1e3: đổi giây → mili giây cho dễ đọc
hold on;% giữ để vẽ chồng bản (c)
% ---------- (c) Lặp lại với mọi θ_k = 0 ----------
x0 = zeros(size(t));% tín hiệu mới, cộng dồn từ 0
for k = 1:9% lặp k = 1..9
x0 = x0 + A(k)*cos(2*pi*k*F*t);% giống trên nhưng bỏ pha (θ_k = 0)
end% kết thúc vòng for
sound(x0/max(abs(x0)), Fs);% nghe bản không pha để so sánh với (a)
plot(t(idx)*1e3, x0(idx), 'r--', 'LineWidth', 1.5); hold off;% vẽ chồng, đỏ nét đứt
grid on; xlabel('t (ms)'); ylabel('x(t)');% lưới + nhãn trục (thời gian tính bằng ms)
legend('(a) có pha \theta_k', '(c) \theta_k = 0');% chú thích; \theta_k → θₖ
title('Trumpet – nốt B, 3 chu kỳ');% tiêu đề hình
% Trả lời (c): DẠNG SÓNG khác hẳn (đỉnh/đáy dời chỗ) nhưng ÂM NGHE gần như không đổi —
% tai người cảm nhận chủ yếu theo BIÊN ĐỘ các họa âm (phổ biên độ giống nhau), rất kém nhạy với pha.

%% ===== BÀI 3 – II. BIẾN ĐỔI FOURIER & LỌC TÍN HIỆU ĐIỆN TIM (ECG) — main.m =====
% Cần file hum3hb.mat (PTN) cùng thư mục: chứa vector điện áp hb và bước lấy mẫu T = 0.002 s
% Hai hàm FourierTransform / IFourierTransform nằm CUỐI script này (nếu GV yêu cầu file riêng thì cắt mỗi hàm ra
% FourierTransform.m và IFourierTransform.m, tên file phải trùng tên hàm)
clc; clear; close all;% dọn màn hình, biến, hình
% ---------- 1. Tín hiệu điện tim và phổ biên độ ----------

[f, Xf] = FourierTransform(t, hb);% gọi hàm tự viết: f = trục tần số (Hz), Xf = phổ X(f) (số phức)

subplot(2,1,1);% ô trên: tín hiệu theo thời gian
plot(t, x, 'linewidth', 1);% vẽ x(t), nét dày 1
grid; axis([0 2.5 -0.5 1.5]);% grid (không đối số) = bật/tắt lưới | axis: t từ 0→2.5 s, biên độ -0.5→1.5
xlabel('t (sec)'); ylabel('x(t)');% nhãn 2 trục
subplot(2,1,2);% ô dưới: phổ biên độ
plot(f, abs(Xf), 'linewidth', 1);% abs(Xf): mô-đun số phức = phổ BIÊN ĐỘ |X(f)|
grid; axis([-150 150 0 0.12]);% chỉ xem dải tần -150 → 150 Hz
xlabel('f (Hz)'); ylabel('|X(f)|');% nhãn 2 trục
% Câu hỏi 1 (gợi ý): x(t) có dạng nhịp tim lặp lại (sóng P–QRS–T) nhưng bị "răng cưa" do nhiễu nhỏ chồng lên.
% Trên phổ: phần năng lượng chính tập trung ở TẦN THẤP quanh 0 Hz; ngoài ra có các thành phần/đỉnh lẻ ở TẦN CAO
% tách khỏi phần chính → đó là nhiễu tần số cao (vd nhiễu điện lưới) cần lọc bỏ.
% ---------- 2. Đáp ứng xung, đáp ứng biên độ và pha của bộ lọc ----------
h = (568*exp(-300*t) - 485*exp(-243*t).*cos(176*t) ...% h(t) của bộ lọc (t ≥ 0); ... = viết tiếp dòng dưới
+ 668*exp(-243*t).*sin(176*t) - 83*exp(-93*t).*cos(285*t) ...% .* : nhân từng phần tử 2 vector cùng cỡ
- 255*exp(-93*t).*sin(285*t));% (dòng cuối của biểu thức h, đóng ngoặc)

figure(2);% cửa sổ hình số 2
subplot(2,2,[1 2]);% lưới 2×2, gộp ô 1 và 2 → 1 ô rộng cả hàng trên

xlabel('t (sec)'); ylabel('h(t)');% nhãn 2 trục

plot(f, abs(Hf), 'k', 'linewidth', 1); grid;% 'k': màu đen

plot(f, angleH, 'k', 'linewidth', 1); grid;% vẽ pha, màu đen, bật lưới
axis([-150 150 -10 10]);% f ∈ [-150,150] Hz, pha ∈ [-10,10] rad
xlabel('f (Hz)'); ylabel('\angle H(f)');% \angle: ký hiệu góc ∠
% Câu hỏi 2 (gợi ý): |H| ≈ 1 ở tần thấp rồi giảm dần về 0 khi |f| tăng → bộ lọc THÔNG THẤP (Butterworth bậc 5:
% 5 cực -300, -243±176j, -93±285j đều có mô-đun ≈ 300 rad/s) → tần số cắt fc ≈ 300/(2π) ≈ 48 Hz,
% cho qua vùng |f| < ~48 Hz, chặn tần cao.
% ---------- 3. Phổ và tín hiệu ra của bộ lọc ----------

figure(3);% cửa sổ hình số 3
subplot(2,1,1);% ô trên: phổ tín hiệu ra

axis([-150 150 0 0.12]);% cùng thang với hình 1 để so sánh
xlabel('f (Hz)'); ylabel('|Y(f)|');% nhãn 2 trục

subplot(2,1,2);% ô dưới: tín hiệu ra theo thời gian
plot(t2, real(y), 'r', 'linewidth', 1); grid;% real(): bỏ phần ảo cực nhỏ (sai số tính toán) trước khi vẽ
axis([0 2.5 -0.5 1.5]);% cùng thang với ô x(t) ở hình 1 để so sánh
xlabel('t (sec)'); ylabel('y(t)');% nhãn 2 trục
% Câu hỏi 3 (gợi ý): so với hình 1, phổ ra mất các thành phần tần cao; y(t) mượt, hết răng cưa nhưng vẫn giữ
% nguyên dạng P–QRS–T và nhịp tim → bộ lọc khử nhiễu tần cao, giữ thông tin hữu ích (có trễ pha nhỏ).
% ===================== HÀM: BIẾN ĐỔI FOURIER (dựa trên fft) =====================
function [f, X] = FourierTransform(t, x)% vào: t (trục thời gian đều), x (mẫu tín hiệu) | ra: f (Hz), X(f)
ns = size(x, 2);% ns: số mẫu tín hiệu (số cột của vector hàng x)
dt = t(2) - t(1);% dt: bước thời gian giữa 2 mẫu
N = 2*ns;% tính FFT với N = 2·ns điểm (đệm thêm 0) → phổ mịn gấp đôi
df = 1/(N*dt);% df: bước tần số của kết quả = 1/(tổng thời lượng N·dt)
xp = zeros(1, N);% vector N số 0 để chứa tín hiệu đã sắp xếp lại + phần đệm 0
nns = sum(t < 0);% nns: số mẫu có t < 0 (đếm số phần tử true)
xp(1:ns-nns) = x(nns+1:ns);% fft coi mẫu đầu là t = 0 → đặt phần t ≥ 0 lên ĐẦU xp
xp(N-nns+1:N) = x(1:nns);% phần t < 0 đặt ở CUỐI xp (fft tuần hoàn: cuối mảng ≡ thời gian âm)
Xf = dt*fft(xp);% fft: biến đổi Fourier rời rạc nhanh; nhân dt để xấp xỉ tích phân ∫x(t)e^(-j2πft)dt
n2 = ceil(N/2);% ceil: làm tròn lên → n2 = chỉ số "giữa" phổ
if n2 == N/2% N chẵn (luôn đúng vì N = 2·ns)
X(1:n2-1) = Xf(n2+2:N);% nửa sau kết quả fft = tần số ÂM → đưa lên đầu
X(n2:N) = Xf(1:n2+1);% nửa đầu = tần số 0 và DƯƠNG → đưa ra sau ⇒ X sắp từ f âm → f dương
f = (-n2+1)*df : df : n2*df;% trục tần số tương ứng (N điểm, cách df)
else% N lẻ (để hàm dùng được tổng quát)
X(1:n2-1) = Xf(n2+1:N);% tần số âm → lên đầu
X(n2:N) = Xf(1:n2);% tần số 0 và dương → ra sau
f = (-n2+1)*df : df : (n2-1)*df;% trục tần số (N điểm)
end% kết thúc if
end% kết thúc hàm FourierTransform
% ===================== HÀM: BIẾN ĐỔI FOURIER NGƯỢC (dựa trên ifft) =====================
function [t, x] = IFourierTransform(f, X)% vào: f (trục tần số đều), X(f) | ra: t (s), x(t)
ns = length(X);% ns: số điểm phổ
df = f(2) - f(1);% df: bước tần số
N = ns;% tính ifft với đúng N = ns điểm
dt = 1/(N*df);% bước thời gian của kết quả
Xp = zeros(1, N);% vector chứa phổ
Xp(1:ns) = X;% chép phổ vào
nns = sum(f < 0);% số điểm có tần số âm
Xpp = zeros(1, N);% vector chứa phổ đã sắp lại theo thứ tự của ifft
Xpp(1:ns-nns) = Xp(nns+1:ns);% ifft cần thứ tự: f = 0 và dương trước...
Xpp(N-nns+1:N) = Xp(1:nns);% ...tần số âm ở cuối
xf = N*df*ifft(Xpp);% ifft rồi nhân N·df để xấp xỉ tích phân ∫X(f)e^(j2πft)df
n2 = ceil(N/2);% chỉ số giữa
if n2 == N/2% N chẵn
x(1:n2-1) = xf(n2+2:N);% phần cuối kết quả ifft = thời gian âm → đưa lên đầu
x(n2:N) = xf(1:n2+1);% phần đầu = t ≥ 0 → ra sau ⇒ x sắp theo t tăng dần
t = (-n2+1)*dt : dt : n2*dt;% trục thời gian tương ứng
else% N lẻ
x(1:n2-1) = xf(n2+1:N);% thời gian âm → lên đầu
x(n2:N) = xf(1:n2);% t ≥ 0 → ra sau
t = (-n2+1)*dt : dt : (n2-1)*dt;% trục thời gian (N điểm)
end% kết thúc if
end% kết thúc hàm IFourierTransform
